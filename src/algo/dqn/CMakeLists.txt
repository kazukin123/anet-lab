file(GLOB_RECURSE ANET_ALGO_SOURCES CONFIGURE_DEPENDS "src/*.cpp" "src/*.hpp" "include/*.hpp")
file(GLOB_RECURSE ANET_ALGO_TESTS CONFIGURE_DEPENDS "src/*_test.cpp")
list(REMOVE_ITEM ANET_ALGO_SOURCES ${ANET_ALGO_TESTS})
list(FILTER ANET_ALGO_SOURCES EXCLUDE REGEX "/src/test/")

add_library(anet-algo-dqn STATIC ${ANET_ALGO_SOURCES})
target_include_directories(anet-algo-dqn PUBLIC include PRIVATE src)
target_link_libraries(anet-algo-dqn PUBLIC anet-core-base)

add_executable(anet-algo-dqn-test ${ANET_ALGO_TESTS})
target_include_directories(anet-algo-dqn-test PRIVATE src)
target_link_libraries(anet-algo-dqn-test PRIVATE anet-algo-dqn anet-core-base anet-core-testing)
anet_set_test_runtime(anet-algo-dqn-test)
add_test(NAME anet-algo-dqn-test COMMAND anet-algo-dqn-test)
set_tests_properties(anet-algo-dqn-test PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
