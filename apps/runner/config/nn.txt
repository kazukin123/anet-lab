# ------------------------------------------------------------------------------
# Structure Definition Samples
#    最終出力は必ず「1次元ベクトル」になるように定義
# ------------------------------------------------------------------------------

# [Pattern 1] Simple MLP
# 入力 -> Flatten -> Dense -> Dense -> (Vector Output)
net.body.MLP.structure = Flatten > Linear_120 > ReLU > Linear_84 > ReLU

# [Pattern 2] 1D-CNN (Atari Style)
# Conv -> Conv -> Flatten -> (Vector Output)
net.body.Conv1D.structure = Conv1d_S > ReLU > Conv1d_M > ReLU > Flatten

# [Pattern 3] 2D-CNN (Image Style)
# Conv -> Conv -> Flatten -> (Vector Output)
net.body.Conv2D.structure = Conv2d_S > ReLU > Conv2d_M > ReLU > Flatten

# [Pattern 4] Transformer
# Linear -> PosEnc -> Transformer -> Pooling -> (Vector Output)
net.body.Transformer.structure = Linear_128 > ReLU > PosEnc > TransEnc_Block > GlobalPool

# [Pattern 5] ResNet-like (Complex Graph)
net.body.ResNet.structure = Linear_128 > ReLU > Linear_128(=skip_val) > Linear_128 > ReLU > Linear_128(=proc_val) > Add(@skip_val @proc_val) > ReLU > Linear_512 > MyReLU

# [Pattern 6] ResNet-152 (Many layers)
net.body.ResNet152.structure = Conv1_Input > MaxPool > ResA(*3) > ResB(*8) > ResC(*36) > ResD(*3) > GlobalAvgPool

## 記法[net.body.structure]
# 記載      意味	              例
# BlockName ブロックの配置        Conv1
#  >		ブロックの連結        Conv1 > ReLU > Conv2
# (=Tag)    出力に名前を付ける    Conv1(=skip)
# (@Source) 入力元をtagで指定する Add(@skip @main)
# (*N)      同じブロックを繰り返し配置する      ResBlock(*3)
