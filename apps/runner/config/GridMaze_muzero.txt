# ---- Representation Network
# 入力: vector [B, 2] (x, y)
# 出力: hidden_state [B, H]

net.rep.branch.[main_branch].bind = vector
net.rep.branch.[main_branch].structure = MuZero_Linear256 > ReLU > MuZero_Linear256 > ReLU

net.rep.body.output.[features] = main_branch


# ---- Dynamics Network
# 入力: hidden_state [B, H], action_one_hot [B, 4] (Up, Down, Left, Right)
# 出力: hidden_state [B, H]
# 出力: reward_feature [B, 128] (reward hidden)

net.dyn.branch.[main_feature].bind = hidden_state, action_one_hot
net.dyn.branch.[main_feature].structure = MuZero_Linear256 > ReLU > MuZero_Linear256 > ReLU

net.dyn.branch.[reward_branch].bind = main_feature
net.dyn.branch.[reward_branch].structure = MuZero_Linear128 > ReLU

net.dyn.body.output.[hidden_state] = main_feature
net.dyn.body.output.[reward_feature] = reward_branch
